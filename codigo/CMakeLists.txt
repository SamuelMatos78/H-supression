add_library(hs_agents STATIC src/engine.c src/policies.c src/cases.c)
target_include_directories(hs_agents PUBLIC include)
target_link_libraries(hs_agents PUBLIC m)
add_executable(hs_arena src/driver.c)
target_link_libraries(hs_arena PRIVATE hs_agents)
add_executable(hs_agent_tests tests/test.c)
target_link_libraries(hs_agent_tests PRIVATE hs_agents)
foreach(t hs_agents hs_arena hs_agent_tests)
    target_compile_options(${t} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    if(HS_SANITIZE)
        target_compile_options(${t} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(${t} PRIVATE -fsanitize=address,undefined)
    endif()
endforeach()
add_test(NAME autonomous_agent_properties COMMAND hs_agent_tests)
